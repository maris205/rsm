def digest(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def compare(group, observed, expected, rtol='1e-58', atol='0', label='', exact=False):
    g = GROUPS.setdefault(group, {'count':0,'passed':0,'max_scaled_error':'0',
                                 'rtol':str(rtol),'atol':str(atol),'exact':exact,
                                 'worst_label':''})
    g['count'] += 1
    if exact:
        ok = str(observed) == str(expected)
        scaled = mp.mpf(0 if ok else 1)
    else:
        ob, ex = mp.mpf(str(observed)),mp.mpf(str(expected))
        delta = abs(ob-ex)
        denom = max(abs(ex),mp.mpf('1e-300'))
        scaled = delta/denom
        ok = bool(delta <= mp.mpf(rtol)*abs(ex)+mp.mpf(atol))
    if scaled > mp.mpf(g['max_scaled_error']):
        g['max_scaled_error'],g['worst_label'] = mp.nstr(scaled,30),str(label)
    g['passed'] += int(ok)
    if not ok:
        FAILURES.append({'group':group,'label':str(label),'observed':str(observed),
                         'expected':str(expected),'scaled_error':mp.nstr(scaled,40),
                         'rtol':str(rtol),'atol':str(atol),'exact':exact})
    return ok

def read_csv(stem):
    with (BASE/'results'/f'{stem}.csv').open(newline='') as f:
        rows = list(csv.DictReader(f))
    ROW_COUNTS[stem]=len(rows)
    CSV_CELLS[stem]=sum(len(r) for r in rows)
    return rows

def cell(stem, index, row, key, val, rtol='1e-58', atol='0', exact=False):
    compare(stem+'.'+key,row[key],val,rtol,atol,f'row {index}',exact)

                  for n in (2,4,8,16,32,64,127,128,160)
                  for r in ('0.1','0.5','0.9','0.99')]

for i,(row,(q,n,r)) in enumerate(zip(phase,expected_phase)):

    W=sum(q**(2*a) for a in range(n+1))
    logmu=mp.log10(mp.mpf(W)/2)/2+(D-2)*mp.log10(mp.mpf(r))
    for key,val in [('q',q),('n',n),('degree_exact',D)]:
        cell('phase_lifting',i,row,key,val,exact=True)
    cell('phase_lifting',i,row,'v_over_cutoff',r)
    cell('phase_lifting',i,row,'log10_degree',mp.log10(D))
    cell('phase_lifting',i,row,'log10_mu_over_v',logmu,atol='1e-58')

abstract_tau=mp.mpf(q)**(1-n)
abstract_zeta=mp.mpf(q*q)*(1-mp.mpf(q)**(-2*n))/(q*q-1)

def dense_spectrum(eta):
    # Independent dense singular values of Q, not eigenvalues of main H.
    Q=np.zeros((n+1,n+1))
    Q[0,0]=q
    for a in range(1,n):
        Q[a,a-1],Q[a,a]=1,-q
    Q[n,n-1],Q[n,n]=1,float(eta)
    singular=np.linalg.svd(Q,compute_uv=False)
    return float(M)*singular[-1],float(M)*singular[0]

gauge=read_csv('gauge_scan')
compare('grid.gauge_rows',len(gauge),59,exact=True)
gap_records=[]
for i,row in enumerate(gauge):
    eta=mp.mpf('0.1')+i*mp.mpf('0.05')
    low,high=dense_spectrum(eta)
    radial1=M/mp.sqrt(2)
    radial3=3*radial1
    gap=min(mp.mpf(str(low)),radial3)
    expected={'q':q,'n':n,'eta':eta,'M_eV':M,'Lambda_eV':Lambda,
              'lightest_vector_eV':low,'heaviest_vector_eV':high,
              'radial_mass_lambda_over_g_1_eV':radial1,
              'radial_mass_lambda_over_g_3_eV':radial3,
              'all_heavy_gap_lambda_over_g_3_eV':gap,
              'probe_over_gap_lambda_over_g_3':probe/gap,
              'gap_criterion_lambda_over_g_3':bool(probe/gap<=mp.mpf('0.1')),
              'abstract_tau':abstract_tau,'physical_local_tau':0,
              'physical_local_zeta':q*q,'abstract_zeta':abstract_zeta,
              'physical_source_h2':eta**-2}
    for key,val in expected.items():
        if key in ('q','n','gap_criterion_lambda_over_g_3','physical_local_tau','physical_local_zeta'):
            cell('gauge_scan',i,row,key,val,exact=True)
        elif key in ('abstract_tau','abstract_zeta'):
            cell('gauge_scan',i,row,key,val)
        else:
            cell('gauge_scan',i,row,key,val,rtol='2e-12')
    gap_records.append({'eta':str(eta),'dense_svd_lightest_vector_eV':str(low),
                        'dense_svd_heaviest_vector_eV':str(high),
                        'all_heavy_gap_eV':mp.nstr(gap,40),
                        'ratio':mp.nstr(probe/gap,40),
                        'passes':expected['gap_criterion_lambda_over_g_3']})

def thomas_link_resolvent(eta,s):
    # L=Q Q^T is tridiagonal in link space; this solves (L+sI)x=e_right.
    # Its first off-diagonal has the opposite sign to the remainder.
    d=[mp.mpf(q*q)+s]+[mp.mpf(q*q+1)+s]*(n-1)+[1+eta*eta+s]
    off=[mp.mpf(q)]+[-mp.mpf(q)]*(n-1)
    rhs=[mp.mpf(0)]*n+[mp.mpf(1)]
    for a in range(1,n+1):
        t=off[a-1]/d[a-1]
        d[a]-=t*off[a-1]
        rhs[a]-=t*rhs[a-1]
    x=[mp.mpf(0)]*(n+1)
    x[n]=rhs[n]/d[n]
    for a in range(n-1,-1,-1):
        x[a]=(rhs[a]-off[a]*x[a+1])/d[a]
    return -s*x[0]/q

momentum=read_csv('momentum_profile')
compare('grid.momentum_rows',len(momentum),393,exact=True)
momentum_max=[]
for i,row in enumerate(momentum):
    ei,si=divmod(i,131)
    eta=(mp.mpf('0.5'),mp.mpf(1),mp.mpf(2))[ei]
    s=mp.power(10,-12+mp.mpf(si)/10)
    cross=thomas_link_resolvent(eta,s)
    expected={'eta':eta,'s':s,'p_over_M':mp.sqrt(s),'local_cross':cross,
              'local_tau':q*q*cross,'ratio_to_abstract_static':q*q*cross/abstract_tau}
    for key,val in expected.items():
        cell('momentum_profile',i,row,key,val)
    momentum_max.append(expected['ratio_to_abstract_static'])

# Numerical algorithm controls independent of the CSV: direct dense inversion
# at small n is safe and checks the signs used in the link-space Thomas solve.
for small_n in (2,5):
    nn=n
    n=small_n
    for eta in (mp.mpf('0.5'),mp.mpf(2)):
        s=mp.mpf('0.125')
        Q=mp.zeros(n+1)
        Q[0,0]=q
        for a in range(1,n):Q[a,a-1],Q[a,a]=1,-q
        Q[n,n-1],Q[n,n]=1,eta
        H=Q.T*Q
        aa=mp.matrix([1]+[0]*n)
        bb=mp.matrix([0]*(n-1)+[1,eta])
        direct=(aa.T*(H+s*mp.eye(n+1))**-1*bb)[0]
        compare('algorithm.thomas_direct',thomas_link_resolvent(eta,s),direct,
                rtol='1e-150',label=f'n{n} eta{eta}')
    n=nn
